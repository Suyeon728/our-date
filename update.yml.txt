name: Refresh places and publish
on:
  workflow_dispatch:
    inputs:
      refresh_places:
        description: 'Rebuild nationwide cafe/library/restaurant database'
        type: boolean
        default: true
  schedule:
    - cron: '23 21 * * *'
  push:
    branches: [main]
    paths-ignore:
      - 'jeonnam-date-v2/places.json'
      - 'jeonnam-date-v2/events.json'
permissions:
  contents: write
  pages: write
  id-token: write
concurrency:
  group: pages
  cancel-in-progress: false
jobs:
  update:
    runs-on: ubuntu-latest
    timeout-minutes: 90
    defaults:
      run:
        working-directory: jeonnam-date-v2
    steps:
      - uses: actions/checkout@v4
        with:
          ref: main
          fetch-depth: 0
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: Initialize event database if missing
        run: test -f events.json || echo '{"events":[]}' > events.json
      - name: Collect nationwide cultural events first
        id: events
        continue-on-error: true
        env:
          DATA_GO_KR_KEY: ${{ secrets.DATA_GO_KR_KEY }}
          SEOUL_API_KEY: ${{ secrets.SEOUL_API_KEY }}
          BUSAN_ENABLED: ${{ vars.BUSAN_ENABLED }}
        run: python collect_events.py
      - name: Save event database
        run: |
          git config user.name github-actions
          git config user.email github-actions@users.noreply.github.com
          git add events.json
          if ! git diff --cached --quiet; then
            git commit -m 'Refresh nationwide cultural events'
            git pull --rebase origin main
            git push origin main
          fi
      - name: Check if nationwide place database needs refresh
        id: places_due
        env:
          RUN_KIND: ${{ github.event_name }}
          FORCE_REFRESH: ${{ inputs.refresh_places }}
        run: |
          python - <<'PYCODE'
          import json, os, datetime
          from pathlib import Path
          try:
              current = json.loads(Path('places.json').read_text())
          except Exception:
              current = {}
          due = current.get('coverage') != 'national-osm' or not current.get('places')
          due = due or (os.environ['RUN_KIND'] == 'schedule' and (datetime.datetime.now(datetime.timezone.utc)+datetime.timedelta(hours=9)).weekday()==0)
          due = due or os.environ.get('FORCE_REFRESH') == 'true'
          with open(os.environ['GITHUB_OUTPUT'],'a') as f: f.write('refresh='+str(due).lower()+'\n')
          PYCODE
      - name: Build nationwide offline place database
        id: nationwide
        if: steps.places_due.outputs.refresh == 'true'
        continue-on-error: true
        timeout-minutes: 35
        run: |
          python -m pip install 'osmium>=4,<5' 'shapely>=2,<3'
          curl --fail --location --retry 3 --connect-timeout 20 --max-time 600 'https://download.geofabrik.de/asia/south-korea-latest.osm.pbf' -o "$RUNNER_TEMP/korea.osm.pbf"
          python collect.py "$RUNNER_TEMP/korea.osm.pbf"
      - name: Save nationwide place database
        if: steps.nationwide.outcome == 'success'
        run: |
          git add places.json
          if ! git diff --cached --quiet; then
            git commit -m 'Refresh national OSM place database'
            git pull --rebase origin main
            git push origin main
          fi
      - name: Report place database failure
        if: steps.nationwide.outcome == 'failure'
        run: echo '::warning::Nationwide place refresh failed. Previous place database retained.'
      - uses: actions/configure-pages@v5
      - name: Prepare website
        run: |
          mkdir -p ../public
          cp index.html places.json events.json manifest.json sw.js icon.png ../public/
      - uses: actions/upload-pages-artifact@v3
        with:
          path: public
      - name: Deploy website
        uses: actions/deploy-pages@v4
      - name: Report event API failure
        if: steps.events.outcome == 'failure'
        run: |
          echo 'Event collection failed or was partial. Check event collection log.'
          exit 1
      - name: Check national place update outcome
        if: always() && steps.nationwide.outcome == 'failure'
        run: |
          echo 'Nationwide place database could not be refreshed; check Build nationwide offline place database.'
          exit 1
